// Problem: Climbing Stairs
// Difficulty: easy
// URL: https://leetcode.com/problems/climbing-stairs/

/* 
Notes:
return func(n-1)+func(n-2)
*/

class Solution {
public:
int dp[46];
    int climbStairs(int n) {
        memset(dp,-1,sizeof(dp));
        return solve(n);
    };
    int solve(int n){
        if(n==1 || n==2 || n==3){
            return n;
        }
        if(dp[n]!=-1){
            return dp[n];
        }
        int a=solve(n-1);
        int b=solve(n-2);
        return dp[n]=a+b;
    }
};
