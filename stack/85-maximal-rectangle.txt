// Problem: Maximal Rectangle
// Difficulty: hard
// URL: https://leetcode.com/problems/maximal-rectangle/

class Solution {
public:
    vector<int> findNSL(vector<int>& height){
        int n=height.size();
        stack<int> st;
        vector<int> NSL(n);
        for(int i=0;i<height.size();i++){
            if(st.empty()){
                NSL[i]=-1;
            }
            else{
                while(!st.empty() && height[st.top()]>=height[i]){
                    st.pop();
                }
                if(st.empty()){
                    NSL[i]=-1;
                }
                else{
                    NSL[i]=st.top();
                }
            }
            st.push(i);
        }
        return NSL;
    }
    vector<int> findNSR(vector<int>& height){
        int n=height.size();
        stack<int> st;
        vector<int> NSR(n);
        for(int i=n-1;i>=0;i--){
            if(st.empty()){
                NSR[i]=n;
            }
            else{
                while(!st.empty() && height[st.top()]>=height[i]){
                    st.pop();
                }
                if(st.empty()){
                    NSR[i]=n;
                }
                else{
                    NSR[i]=st.top();
                }
            }
            st.push(i);
        }
        return NSR;
    }
    int findMaxArea(vector<int>& height){
        int n=height.size();
        vector<int> NSL=findNSL(height);
        vector<int> NSR=findNSR(height);
        vector<int> width(n);
        for(int i=0;i<n;i++){
            width[i]=NSR[i]-NSL[i]-1;
        }
        int maxArea=0;
        for(int i=0;i<n;i++){
            maxArea=max(maxArea,height[i]*width[i]);
        }
        return maxArea;
    }
    int maximalRectangle(vector<vector<char>>& matrix) {
        int n=matrix.size();
        int m=matrix[0].size();
        vector<int> height(m,0);
        for(int i=0;i<m;i++){
            height[i]+=(matrix[0][i]=='0'?0:1);
        }
        int maxArea=findMaxArea(height);
        for(int row=1;row<n;row++){
            for(int col=0;col<m;col++){
                if(matrix[row][col]=='1'){
                    height[col]+=1;
                }
                else{
                    height[col]=0;
                }
            }
            maxArea=max(maxArea,findMaxArea(height));
        }
        return maxArea;
    }
};
